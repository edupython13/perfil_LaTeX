% 7. CUSTOM COMMANDS (OPTIMIZED AND ROBUST)

% Simple image inclusion
\newcommand{\pdf}[3]{%
	\begin{center}%
		\includegraphics[scale=#1, page=#2]{#3}%
	\end{center}%
}

% Image inclusion with optional trimming
% #1: scale, #2: page, #3: file, #4: trim values, #5: trim flag (true/false)
\newcommand{\pdft}[5]{%
	\begin{center}%
		\ifthenelse{\equal{#5}{true}}{%
			\includegraphics[scale=#1, page=#2, trim=#4, clip]{#3}%
		}{%
			\includegraphics[scale=#1, page=#2]{#3}%
		}%
	\end{center}%
}

% Citation command for author-year style
\newcommand{\xcite}[1]{(\citeauthor{#1}, \citeyear{#1})}

% Counter for left-aligned paragraphs in TOC
\newcounter{tocleftparcounter}

% Command to add left-aligned paragraph to TOC
\makeatletter
\newcommand{\tocleftparagraph}[1]{%
	\stepcounter{tocleftparcounter}%
	\edef\tocleftparlabel{tocleftpar.\thetocleftparcounter}%
	\hypertarget{\tocleftparlabel}{}%
	\addtocontents{toc}{%
		\protect\normalsize\protect\l@section{%
			\protect\hyperlink{\tocleftparlabel}{%
				\protect\RaggedRight\protect\textbf{#1}\protect\cftsecleader%
			}%
		}{%
			\protect\hyperlink{\tocleftparlabel}{%
				\protect\cftsecpagefont\protect\textbf{\thepage}%
			}%
		}%
	}%
}
\makeatother

% ===================================================================
% 10. LANDSCAPE PAGE STYLE - Versión por Coordenadas Absolutas
% ===================================================================
\newif\iflandscapemode
\landscapemodefalse

\makeatletter
% Estilo vacío para anular el número estándar que pone longtable
\def\ps@landscapeplain{%
  	\let\@mkboth\@gobbletwo
  	\let\@oddhead\@empty
  	\let\@evenhead\@empty
  	\let\@oddfoot\@empty
  	\let\@evenfoot\@empty
}

\let\original@ps@plain\ps@plain

\AddToShipoutPictureBG{%
  	\iflandscapemode
  	  	\begin{tikzpicture}[remember picture, overlay]
  	  		% REQUERIMIENTO:
  	  		% 1.5 cm desde el borde inferior visual (que es el borde derecho físico)
  	  		% 3.0 cm desde el borde derecho visual (que es el borde superior físico)
  	  		% Usamos 'current page.south west' como origen (0,0) para mayor control.
  	  		% En una hoja Letter (21.59 x 27.94 cm) rotada:
  	  		\node[rotate=90, anchor=east, font=\normalsize] at ([xshift=20.09cm, yshift=24.94cm]current page.south west) {\thepage};
  	  		% Explicación de las coordenadas:
  	  		% x = 21.59cm (ancho) - 1.5cm = 20.09cm
  	  		% y = 27.94cm (alto) - 3.0cm = 24.94cm
  	  	\end{tikzpicture}%
  	\fi
}

\AtBeginEnvironment{landscape}{%
  	\clearpage
  	\global\landscapemodetrue
  	\let\ps@plain\ps@landscapeplain
  	\pagestyle{landscapeplain}
}

\AtEndEnvironment{landscape}{%
  	\clearpage
  	\global\landscapemodefalse
  	\let\ps@plain\original@ps@plain
  	\pagestyle{plain}
}
\makeatother
