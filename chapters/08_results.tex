% ===================================================================
% MACRO FOR CONDITIONAL TABLE INPUT
% ===================================================================
% This command checks if a table file exists in the 'tables/' directory.
% If it exists, it starts a new page and inputs the table.
% If not, it outputs a warning to the console.
% Improves maintainability by centralizing file existence checks.
\newcommand{\inputtable}[1]{%
    \IfFileExists{tables/#1.tex}{%
        \clearpage%
        \input{tables/#1}%
    }{%
        \typeout{Warning: Table file tables/#1.tex not found}%
    }%
}
